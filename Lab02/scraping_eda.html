BLOGS = {
    'TechnCruncher': 'https://techncruncher.blogspot.com',
    'BloggerBuzz':   'https://blogger.googleblog.com',   # <- change if this blog has < 10 usable posts
}
PER_BLOG = 10
HEADERS = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)'}

def scrape_blog(name, base, n=PER_BLOG):
    rows = []
    feed = f'{base}/feeds/posts/default?alt=json&max-results=50'
    data = requests.get(feed, headers=HEADERS, timeout=15).json()
    for e in data['feed'].get('entry', []):
        html = e.get('content', {}).get('$t', '')
        soup = BeautifulSoup(html, 'lxml')
        paras = [p.get_text(' ', strip=True) for p in soup.find_all('p')]
        text = ' '.join(paras) if paras else soup.get_text(' ', strip=True)
        text = ' '.join(text.split())
        if len(text) < 300:          # quality filter: skip very short posts
            continue
        url = next((l['href'] for l in e['link'] if l['rel'] == 'alternate'), '')
        rows.append({'blog': name, 'url': url, 'title': e['title']['$t'].strip(), 'text': text})
        if len(rows) == n:
            break
    time.sleep(1)                    # be polite
    return rows

articles = []
for name, base in BLOGS.items():
    got = scrape_blog(name, base)
    print(name, '->', len(got), 'articles')
    articles += got
df = pd.DataFrame(articles).drop_duplicates('url').reset_index(drop=True)
assert df.groupby('blog').size().min() >= 10, 'Need 10 articles per blog: change BLOGS or lower the length filter'

def spacy_features(text):
    doc = nlp(text)
    n_tok = len(doc)
    n_noun = sum(t.pos_ in ('NOUN', 'PROPN') for t in doc)   # common + proper nouns
    return pd.Series({'num_tokens': n_tok,
                      'num_sentences': len(list(doc.sents)),
                      'num_entities': len(doc.ents),
                      'num_nouns': n_noun,
                      'noun_density': n_noun / n_tok if n_tok else 0})

def sentiment(text):
    s = TextBlob(text).sentiment
    return pd.Series({'sentiment_polarity': s.polarity, 'sentiment_subjectivity': s.subjectivity})

df = pd.concat([df, df['text'].apply(spacy_features), df['text'].apply(sentiment)], axis=1)
df['title_length'] = df['title'].str.len()
df.to_csv('articles.csv', index=False)
df.drop(columns='text').head()
